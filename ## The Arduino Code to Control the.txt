## The Arduino Code to Control the Servo Motors

// C++ code
//
#include <Servo.h>

Servo m1;
Servo m2;
Servo m3;
Servo m4;

void setup() {
  m1.attach(3);
  m2.attach(5);
  m3.attach(6);
  m4.attach(9);

  unsigned long startTime = millis();

  while (millis() - startTime < 2000) {

    for (int pos = 0; pos <= 180; pos++) {
      m1.write(pos);
      m2.write(pos);
      m3.write(pos);
      m4.write(pos);
      delay(15);

      if (millis() - startTime >= 2000)
        break;
    }

    for (int pos = 180; pos >= 0; pos--) {
      m1.write(pos);
      m2.write(pos);
      m3.write(pos);
      m4.write(pos);
      delay(15);

      if (millis() - startTime >= 2000)
        break;
    }
  }

  m1.write(90);
  m2.write(90);
  m3.write(90);
  m4.write(90);
}

void loop() {
}
